 """
 QHSJ AIGC 全能影视 —— 全维导演创世与双区生成总引擎
 永久优先级：

 第一优先级：导演思维（QHSJ永久底层 + QHSJ导演插件）
 第二优先级：空间推演
 第三优先级：反推引擎
 第四优先级：模型适配
 第五优先级：双区输出
 第六优先级：质量回归

 def __init__(self):
 self.QHSJ内核 = QHSJ 永久导演思维内核()
 self.导演插件 = QHSJ导演思维插件库()
 self.空间推演 = 空间推演引擎()
 self.反推 = 反推还原引擎()
 self.模型适配 = 模型适配引擎()
 self.输出 = 双区输出引擎()
 self.校验 = 自动校验引擎()
 self.修正 = 自动修正引擎()
 self.质量回归 = QHSJ质量回归引擎()
 self.审美仲裁 = 混沌审美仲裁()
 def 运行(self, 用户输入):
 # 第一优先级：导演思维
 QHSJ方案 = self.QHSJ内核.自动运行(用户输入)
 if 用户输入.指定导演:

 插件 = self.导演插件.加载(用户输入.指定导演)
 elif 用户输入.题材或情绪:
 插件 = self.导演插件.自动匹配(用户输入)
 else:
 插件 = None
 if 插件:
 导演方案 = 插件.融合(QHSJ方案, 用户输入)
 导演方案 = 冲突仲裁.运行(导演方案, 插件)
 else:
 导演方案 = QHSJ方案
 # 第二优先级：空间推演
 全局空间 = self.空间推演.全局建模(用户输入)
 物体布局 = self.空间推演.物体布局图(全局空间)
 距离表 = self.空间推演.距离表(全局空间)
 角色锚点 = self.空间推演.角色锚点表(用户输入)
 坐标推演 = self.空间推演.三维坐标推演(全局空间)
 空间表 = self.空间推演.空间表(全局空间, 导演方案)
 空间图 = self.空间推演.空间图(空间表)
 沙盘 = self.空间推演.沙盘推演(空间表, 空间图)
 轴线 = self.空间推演.轴线锁定(空间表)
 衔接 = self.空间推演.前后镜衔接(空间表)
 终帧 = self.空间推演.终帧锁死(空间表)

 画面语言 = self.空间推演.转译(空间表, 空间图, 沙盘)
 # 第三优先级：反推引擎
 if 用户输入.有参考图 or 用户输入.有参考片:
 拆图 = self.反推.七要素拆图(用户输入.参考)
 模式 = self.反推.判断模式(用户输入)
 if 模式 == "1比1还原":
 反推结果 = self.反推.完全还原(拆图)
 else:
 反推结果 = self.反推.风格抽取(拆图)
 else:
 反推结果 = None
 # 第四优先级：模型适配
 模型 = self.模型适配.询问模型(用户输入)
 if 模型 == "未知":
 模型 = self.模型适配.联网查询(用户输入)
 适配格式 = self.模型适配.校准格式(模型)
 适配提示词 = self.模型适配.转换格式(画面语言, 适配格式)
 # 第五优先级：双区输出
 框外 = self.输出.框外方案(
 导演方案, 全局空间, 空间表, 空间图,
 沙盘, 轴线, 衔接, 终帧, 反推结果
 )

 框内 = self.输出.框内提示词(
 适配提示词, 角色锚点, 物理锁定,
 场景锚点, 负向词
 )
 # 自动校验与修正
 校验结果 = self.校验.运行(框外, 框内)
 if 校验结果.有矛盾:
 框外, 框内 = self.修正.运行(框外, 框内, 校验结果)
 # 第六优先级：质量回归
 质量报告 = self.质量回归.运行(框外, 框内)
 审美仲裁结果 = self.审美仲裁.运行(质量报告)
 经验沉淀 = self.质量回归.沉淀(质量报告, 审美仲裁结果)
 return {
 "框外": 框外, "框内": 框内, "导演方案": 导演方案,
 "空间模型": 全局空间, "反推结果": 反推结果,
 "模型适配": 模型, "校验报告": 校验结果,
 "质量报告": 质量报告, "经验沉淀": 经验沉淀
 }

1.2 移动状态规则
- 每镜必须写移动方式、速度、身体姿态、步频、步幅、手臂摆动、脚掌动作、呼吸状态、环境反馈、
是否全程不停
- 前后镜号移动状态保持一致或合理过渡

- 追赶场景必须写"极速狂奔"或"奔跑"或"冲刺"，不能只写"追"
- 逃窜场景必须写"极速狂奔"或"奔跑"或"小跑"，不能只写"逃"
- 回头看、说话时必须是极速狂奔状态，不能停下来
- 移动状态与镜头时长匹配，极速狂奔镜头至少2秒，冲刺镜头至少2.5秒
- 镜头内移动状态全程不停，没有慢下来，没有停下来
